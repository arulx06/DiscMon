import json
import subprocess
import time

import psycopg


def command(*args):
    print('$ ' + ' '.join(args), flush=True)
    result = subprocess.run(args, capture_output=True, text=True, check=True)
    print(result.stdout.strip(), flush=True)
    if result.stderr:
        print(result.stderr.strip(), flush=True)
    return result.stdout.strip()


def wait_until(predicate):
    deadline = time.monotonic() + 15
    while not predicate():
        if time.monotonic() >= deadline:
            raise TimeoutError('isolated test readiness condition')
        time.sleep(0.02)


dsn = 'postgresql://discmon:discmon@127.0.0.1:55439/discmon_task17c_container_test'
with psycopg.connect(dsn, autocommit=True) as observer:
    for mode in ('idle', 'database-lock'):
        name = f'discmon-task17c-scheduler-{mode}'
        holder = psycopg.connect(dsn)
        if mode == 'database-lock':
            holder.execute('LOCK TABLE jobs IN ACCESS EXCLUSIVE MODE')
        command('docker', 'run', '-d', '--name', name,
                '--network', 'discmon-task17c-isolated',
                '-e', 'APP_ENV=test', '-e', 'SHEETS_SYNC_ENABLED=false',
                '-e', 'DATABASE_URL=postgresql+psycopg://discmon:discmon@discmon-task17c-postgres:5432/discmon_task17c_container_test?application_name=task17c-docker',
                'discmon-task17c-api', 'python', '-m', 'app.scheduler')
        def database_state():
            return observer.execute(
                "SELECT state, wait_event_type FROM pg_stat_activity "
                "WHERE application_name = 'task17c-docker'"
            ).fetchall()
        if mode == 'idle':
            wait_until(lambda: any(state == 'idle' for state, _ in database_state()))
        else:
            wait_until(lambda: any(wait == 'Lock' for _, wait in database_state()))
        print('Observed DB state:', database_state(), flush=True)
        started = time.monotonic()
        print('$ docker stop --time=-1 ' + name, flush=True)
        stop = subprocess.Popen(['docker', 'stop', '--time=-1', name],
                                stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True)
        if mode == 'database-lock':
            time.sleep(0.3)
            assert stop.poll() is None
            assert any(wait == 'Lock' for _, wait in database_state())
            print('Still alive after SIGTERM with transaction waiting; releasing test lock', flush=True)
            holder.commit()
        stdout, stderr = stop.communicate(timeout=15)
        print(stdout.strip(), stderr.strip(), flush=True)
        print(f'mode={mode} unlimited-stop elapsed={time.monotonic()-started:.3f}s CLI-exit={stop.returncode}', flush=True)
        assert stop.returncode == 0
        state = json.loads(command('docker', 'inspect', '--format', '{{json .State}}', name))
        assert state['ExitCode'] == 0 and not state['OOMKilled'] and not state['Running']
        logs = command('docker', 'logs', name)
        assert 'scheduler_stopped' in logs
        assert 'scheduler_error' not in logs
        print('queued jobs:', observer.execute('SELECT count(*) FROM jobs WHERE state = %s', ('queued',)).fetchone()[0], flush=True)
        holder.close()
        command('docker', 'rm', name)
