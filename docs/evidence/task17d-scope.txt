import asyncio
import dis
import signal
import subprocess
import types

from app import worker

original = types.ModuleType('baseline_worker')
exec(subprocess.check_output(['git', 'show', '4e0448f:services/api/app/worker.py'], text=True), original.__dict__)
original._install_signal_handlers()
print('Original callback:')
dis.dis(signal.getsignal(signal.SIGTERM))
signal.raise_signal(signal.SIGTERM)
print('Original actual loop flag after SIGTERM:', original._running)
assert original._running

async def corrected():
    wakeup = asyncio.Event()
    worker._install_signal_handlers(wakeup)
    callback = signal.getsignal(signal.SIGTERM)
    print('Corrected OS callback:')
    dis.dis(callback)
    print('Loop callback:')
    stop = dict(zip(callback.__code__.co_freevars, callback.__closure__))['stop'].cell_contents
    dis.dis(stop)
    signal.raise_signal(signal.SIGTERM)
    print('Before loop observation:', worker._running, wakeup.is_set())
    assert worker._running and not wakeup.is_set()
    await asyncio.sleep(0)
    print('After loop observation:', worker._running, wakeup.is_set())
    assert not worker._running and wakeup.is_set()

asyncio.run(corrected())
